\section{Lecture Summary}
\begin{theorem}
There exists an algorithm TS solving the leader election problem in time $n u_{\min}$ and with communication complexity $O(n)$ for every system $A$ of $n$ processes arranged in a ring, where $n \in \mathbb{N}, n > 1$, is known to the algorithm.
\end{theorem}


\begin{algorithm}
\textbf{Algorithm TS (Time Slice) -} Initially, each process $i$ (for $i = 1, \ldots, n$) initializes a token carrying its own UID.

The algorithm proceeds in phases $1, 2, 3, \ldots$, where each phase consists of exactly $n$ consecutive rounds. In each phase $l$, only the token with UID $l$ is permitted to circulate around the ring.

\textbf{Phase $l$ (for $l = 1, 2, 3, \ldots$):} 
This phase comprises rounds $(l-1)n + 1$ through $ln$. The following happens:
\begin{itemize}
\item If a process $i$ with UID equal to $l$ exists and has not previously received any token with a non-null message by the start of round $(l-1)n + 1$, then process $i$ declares itself the leader.
\item Process $i$ then sends its token all the way around the ring during this phase (one hop per round).
\item Tokens with UIDs other than $l$ are blocked and do not propagate during phase $l$.
\item When the token from process $i$ completes its circulation and returns, process $i$ confirms its leadership.
\end{itemize}

The algorithm terminates when a process declares itself leader.
\end{algorithm}


\begin{theorem}
There exists an algorithm VS solving the leader election problem in time $n 2^{u_{\min}}$ and with communication complexity $O(n)$ for every system $A$ of $n$ processes arranged in a ring, where $n \in \mathbb{N}, n > 1$, and $n$ is not known to the algorithm.
\end{theorem}


\begin{algorithm}
\textbf{Algorithm VS (Variable Speed) -} Each process $i$ (for $i = 1, \ldots, n$) initializes a token carrying its UID $u_i$. 

A token with UID $v$ circulates at variable speed: it advances at a rate of one message transmission every $2^v$ rounds. That is, each process receiving a token with UID $v$ waits $2^v$ rounds before forwarding it to its clockwise neighbor.

\textbf{Token handling:}
\begin{itemize}
\item When a process receives a token, it compares the UID on the token with its own UID.
\item If the UIDs match, the process declares itself the leader.
\item Each process maintains a record of the minimum UID it has seen.
\item Any token with a UID larger than the current minimum is deleted (not forwarded).
\end{itemize}

This approach ensures that eventually, only the token with the globally minimum UID circulates, and the process with that UID will recognize it and become leader.
\end{algorithm}

\begin{lemma}
Let $n \in \mathbb{N}, n > 1$, let $A$ be a comparison-based algorithm executing in a ring $R$ of size $n$. Furthermore, let $i, j$ be any two processes in $A$ and let $k \in \mathbb{N}$ be any number satisfying $0 \leq k \leq \lceil n/2 \rceil$ such that both processes $i, j$ have order equivalent sequences of UIDs in their $k$-neighborhoods. \\
Then, at any point after at most $k$ active rounds, process $i$ and process $j$ are in corresponding states with respect to the UID sequences in their $k$-neighborhoods.
\end{lemma}
\end{samepage}


\begin{theorem}
There exists a constant $c$ such that, for all $n \in \mathbb{N}, n > 1$, there is a $c$-symmetric ring of size $n$.
\end{theorem}


\begin{lemma}
Let $n \in \mathbb{N}, n > 1$ and let $A$ be a comparison-based algorithm executing in a $c$-symmetric ring $R$ of size $n$. Furthermore, assume that $A$ elects a leader. Suppose $k \in \mathbb{N}$ such that $\sqrt{n} \leq 2k + 1$ and $\lceil cn/(2k + 1) \rceil \geq 2$. Then $A$ has more than $k$ active rounds.
\end{lemma}


\begin{theorem}
Let $n \in \mathbb{N}, n > 1$, and let $A$ be a comparison-based algorithm that elects a leader in rings of size $n$. Then there is an execution of $A$ in which $A$ sends $\Omega(n \log{n})$ messages by the time the leader is elected.
\end{theorem}



\section{Exercises}
\noindent \textbf{Exercise 15.1}  Every bit-reversal ring is $\frac{1}{2}$-symmetric.
\begin{proof}
A bit-reversal ring is a ring of size $n = 2^k$ for some $k \geq 1$, where processes are labeled $0, 1, \ldots, 2^k - 1$ in binary. The UID of process $i$ is $\operatorname{bitreverse}(i)$ (the binary representation of $i$ with bits reversed).

A ring is $c$-symmetric if for all segment lengths $l$ with $\sqrt{n} \leq l \leq n$, there are at least $\lceil cn/l \rceil$ segments of length $l$ that are pairwise order-equivalent. We need to show that $c = 1/2$ works.

\textbf{Base case:} For $l = n/2$, we need at least $\lceil n/(2l) \rceil = \lceil n/(n) \rceil = 1$ segment of length $n/2$. This is trivially satisfied.

\textbf{Key observation for bit-reversal rings:} Consider a segment of length $2^{k-1} = n/2$ starting at position $i$. The relative order of UIDs in this segment depends only on the relative bit-reversal pattern within those positions.

\textbf{Example for $n = 8$ (where $k = 3$):} Consider intervals of length $4$:
\begin{center}
\begin{tabular}{cc}
\textbf{Positions} & \textbf{UIDs (bit-reversed)} \\
$\{0, 4, 2, 6\}$ & $0, 4, 2, 6$ \\
$\{4, 2, 6, 1\}$ & $4, 2, 6, 1$ \\
$\{2, 6, 1, 5\}$ & $2, 6, 1, 5$ \\
$\{6, 1, 5, 3\}$ & $6, 1, 5, 3$ \\
$\{1, 5, 3, 7\}$ & $0, 2, 1, 3$ \\
$\{5, 3, 7, 0\}$ & $2, 1, 3, 0$ \\
$\{3, 7, 0, 4\}$ & $1, 3, 0, 2$ \\
$\{7, 0, 4, 2\}$ & $3, 0, 2, 1$
\end{tabular}
\end{center}

\textbf{Pattern:} Due to the cyclic symmetry of the ring and the bit-reversal operation, for every segment of length $l$ starting at position $i$, there exists a corresponding segment starting at position $(i + n/2) \bmod n$ that has a related order pattern. This doubling of equivalent segments continues at each scale.

\textbf{Inductive argument:} 
\begin{enumerate}
\item For length $l = n/2$: All relative orders in a segment of length $n/2$ can be recovered by bit-reversing values modulo $2^{k-1}$.
\item For length $l < n/2$: By the self-similar structure of bit-reversal rings, if a segment of length $l$ appears $m$ times with order-equivalent arrangements in a ring of size $n/2$, then it appears at least $2m$ times (with order-equivalent arrangements) in the full ring of size $n$.
\item The recursion shows that at level $l$, there are at least $\lceil n/(2l) \rceil$ segments of length $l$ that are order-equivalent.
\end{enumerate}

Therefore, every bit-reversal ring satisfies the $\frac{1}{2}$-symmetry property.
\end{proof}
