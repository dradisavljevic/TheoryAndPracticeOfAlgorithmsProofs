\section{Lecture Summary}
\begin{definition}
The value of an AS-number $x = (x_{n-1}, \ldots, x_{0}) \in \{ 0, 1 \}^{n}$ is defined as
$$V_{AS}(x) =_{df} (-1)^{x_{n-1}}(x_{n-2}2^{n-2} + \cdots + x_{1}2 + x_{0}).$$
\end{definition}


\begin{theorem}
There is a log-space uniform family $(C_{n})_{n \in \mathbb{N}}$ of Boolean circuits achieving simultaneously depth $O(\log{n})$ and size $O(n)$ such that $C_{n}$, on input two $n$-bit AS-numbers $x$, $y$, correctly computes $x + y$.
\end{theorem}


\begin{theorem}
There is no family $(C_{n})_{n \in \mathbb{N}}$ of Boolean circuits over the basis $\{ \lnot, \land, \lor \}$ for binary addition such that $C_{n}$ computes the sum of any two $n$-bit numbers, which achieves depth $o(\log{n})$.
\end{theorem}


\begin{definition}
\textbf{Fermat Ring -} Let $m \in \mathbb{N} \backslash \{ 0 \}$. \\
(i) $\mathbb{Z}_{m}$ is called a Fermat ring provided $m = 4^{k} + 1$ for some $k \in \mathbb{N}, k \geq 1$. \\
(ii) Let $\mathbb{Z}_{m}$ be a Fermat ring and let $x \in \mathbb{Z}_{m}$. A radix-$4$ representation for $x$ is any vector $x = (x_{n}, \ldots, x_{0})$ with $-3 \leq x_{i} \leq 3$ such that 
$$|x|_{4} = x_{n}4^{n} + x_{n - 1}4^{n - 1} + \cdots + x_{0} \equiv x \pmod{m}.$$
\end{definition}


\begin{theorem}
There exists a log-space uniform family $(C_{n})_{n \in \mathbb{N}}$ of Boolean circuits achieving simultaneously depth $O(1)$ and size $O(n)$ such that $C_{n}$, on input any two $3(n + 1)$ bit radix-$4$ numbers $x$, $y$ correctly computes $x + y$ mod $m$, where $m = 4^{n} + 1$.
\end{theorem}


\begin{corollary}
There exists a log-space uniform family $(C_{n})_{n \in \mathbb{N}}$ of Boolean circuits achieving simultaneously depth $O(1)$ and size $O(n)$ such that $C_{n}$, on input two $3(n + 1)$ bit radix-$4$ numbers $x$, $y$ correctly computes $x - y$ mod $m$, where $m = 4^{n} + 1$.
\end{corollary}


\begin{samepage}
\begin{theorem}
(1) Let $a_{x} = (x_{2n}, \ldots, x_{0})$ be the binary representation of a number $a \in \{ 0, \ldots, 2^{2n+1} - 1 \}$, and let $m = 4^{n + 1} + 1$. Then $a_{x}$ can be transformed into a radix-$4$ representation of $a$ by a log-space uniform family $(C_{n})_{n \in \mathbb{N}}$ of Boolean circuits that achieves simultaneously constant depth and linear size. \\
(2) There is a log-space uniform family $(C_{n})_{n \in \mathbb{N}}$ of Boolean circuits that achieves simultaneously depth $O(\log{n})$ and size $O(n)$ such that $C_{n}$ computes on input a radix-$4$ representation $(x_{n}, \ldots, x_{0})$ of any number $a \in \{ 0, \ldots, m - 1 \}$ the binary representation for $a$.
\end{theorem}
\end{samepage}


\begin{theorem}
There exists a log-space uniform family $(C_{n})_{n \in \mathbb{N}}$ of Boolean circuits achieving simultaneously depth $O(\log{n})$ and size $O(n^{2})$ such that $C_{n}$, on input $n$ numbers $a_{1}, \ldots, a_{n}$ having at most $n$ bits each, correctly computes $a_{1} + \cdots + a_{n}$.
\end{theorem}



\section{Exercises}
\noindent \textbf{Exercise 13.1}  Given $m = 4^2 + 1 = 17$, convert the radix-4 representation $(0, 3, 1)$ to binary form and verify the computation.
\begin{proof}
Let $m = 4^{2} + 1 = 17$ and let the input radix-4 representation be $(x_2, x_1, x_0) = (0, 3, 1)$ where each $x_i \in \{-3, \ldots, 3\}$.

We compute the value represented by this radix-4 encoding:
$$|x|_4 = x_2 \cdot 4^2 + x_1 \cdot 4^1 + x_0 \cdot 4^0 = 0 \cdot 16 + 3 \cdot 4 + 1 \cdot 1 = 0 + 12 + 1 = 13$$

Therefore, $|x|_4 \equiv 13 \pmod{17}$. Since $13 < 17 = m$, we have $13 \equiv 13 \pmod{17}$, so the value is indeed $13$.

Now we convert this to binary representation. We need to express $13$ in binary form:
$$13 = 8 + 4 + 1 = 2^3 + 2^2 + 2^0$$

Therefore, the binary representation is $(1, 1, 0, 1)_2$ (reading the bits from most significant to least significant).

Verification: Using the AS-number definition with $x = (x_3, x_2, x_1, x_0) = (1, 1, 0, 1)$ where $x_3$ is the sign bit:
$$V_{AS}(x) = (-1)^{x_3}(x_2 \cdot 2^2 + x_1 \cdot 2 + x_0) = (-1)^1(1 \cdot 4 + 0 \cdot 2 + 1) = -5$$

This represents $-5$. However, for the direct binary to radix-4 conversion (without AS-number), we compute:
$$13 \text{ in binary} = (0, 1, 1, 0, 1)_2$$

Reading as a standard binary number from left to right (MSB first), this is $1 \cdot 8 + 1 \cdot 4 + 0 \cdot 2 + 1 = 13$.
\end{proof}

\noindent \textbf{Exercise 13.2} Prove that there is a uniform family $(C_{n})_{n \in \mathbb{N}}$ of Boolean circuits simultaneously achieving depth $O(1)$ and size $O(n)$ which on input any number $b$ given in a radix-4 representation $(x_{n}, \ldots, x_{0})$ computes the radix-4 representation of the product $b \cdot 2^{l}$, where $l \leq 2n$ is any fixed constant.
\begin{proof}
We show that multiplication by a fixed power of 2 in radix-4 representation can be done with constant depth and linear size.

\textbf{Key observation:} In radix-4 representation, multiplying by $2^l$ shifts the representation by $l$ positions, analogous to how multiplying by 10 shifts a decimal representation.

More precisely, if $b = \sum_{i=0}^{n} x_i 4^i$, then:
$$b \cdot 2^l = \sum_{i=0}^{n} x_i 4^i \cdot 2^l$$

Since $4^i = (2^2)^i = 2^{2i}$, we have:
$$b \cdot 2^l = \sum_{i=0}^{n} x_i 2^{2i+l}$$

\textbf{Radix-4 representation of $b \cdot 2^l$:}

We can rewrite this as:
$$b \cdot 2^l = \sum_{j=0}^{n+\lceil l/2 \rceil} y_j 4^j$$

where the coefficients $y_j$ are obtained by:
\begin{enumerate}
\item If $l$ is even ($l = 2m$): $y_j = x_{j-m}$ for $j \geq m$ and $y_j = 0$ for $j < m$.
\item If $l$ is odd ($l = 2m+1$): We need to compute $x_i 2^l = x_i 2^{2m+1} = x_i \cdot 2 \cdot 4^m$. This requires computing $2 x_i$ (multiplication by 2) and then shifting by $m$ positions in radix-4. The multiplication by 2 can be done without carries for each digit independently in constant time.
\end{enumerate}

\textbf{Circuit construction:}

The circuit performs the following steps:
\begin{itemize}
\item Read the input radix-4 digits $(x_n, \ldots, x_0)$
\item If $l$ is even: Output $(x_{n-m}, \ldots, x_0, 0, \ldots, 0)$ (with $m$ zeros appended)
\item If $l$ is odd: Multiply each input digit by 2, then output in shifted positions
\end{itemize}

Both operations involve:
- Reading input: O(n) size, O(1) depth (parallel reading)
- Shifting/rearranging: O(n) size, O(1) depth (direct rewiring)
- Multiplication by 2 (if needed): O(n) size, O(1) depth (local operation at each digit)

Therefore, the entire circuit has size $O(n)$ and depth $O(1)$.
\end{proof}
