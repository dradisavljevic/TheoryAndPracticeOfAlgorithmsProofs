\section{Lecture Summary}
\begin{theorem}
Every element in a field $\mathbb{F}$ of order $q$ satisfies the equation $x^{q^{n}} - x = 0$ for every $n \in \mathbb{N}^{+}$.
\end{theorem}


\begin{theorem}
Every irreducible polynomial of degree $d$ over a field of order $q$ divides $x^{q^{k}} - x$ if $k$ is any multiple of $d$.
\end{theorem}


\begin{theorem}
In a field of order $q$, the polynomial $x^{q^{k}} - x$ factors into the product of all monic irreducible polynomials whose degrees divide $k$.
\end{theorem}


\begin{theorem}
If $f(x)$ is an irreducible polynomial of degree $m$ over $\mathbb{Z}_{p}$ and $m$ divides $k \in \mathbb{N}^{+}$, then a field of order $p^{k}$ must contain $m$ roots of $f(x)$.
\end{theorem}


\begin{theorem}
If $p$ is a prime and $k \in \mathbb{N}^{+}$ then there exists a unique finite field of order $p^{k}$.
\end{theorem}


\begin{theorem}
Let $n \in \mathbb{N}, n \geq 2$, and $a \in \mathbb{N}^{+}$ be such that $gcd(n, a) = 1$. Then we have the following:
\begin{center}
$n$ is prime if and only if $(X + a)^{n} \equiv (X^{n} + a)$ mod $n$
\end{center}
\end{theorem}


\begin{theorem}
Modular exponentiation can be computed in time \\ $O(max\{\log{a}, \log{m}, \log{x} \}^{3})$.
\end{theorem}


\begin{definition}
\textbf{Pseudo Primes -} Let $n \in \mathbb{N}$ be an odd composite number, and let $b \in \mathbb{N}$ such that $gcd(b, n) = 1$. Then $n$ is said to be pseudo-prime to the base $b$ if $b^{n-1} \equiv 1$ mod $n$.
\end{definition}


\begin{samepage}
\begin{theorem}
Let $n \in \mathbb{N}$ be an odd composite number. Then we have \\
(1) $n$ is pseudo-prime to the base $b$ with gcd($b,n$) = $1$ if and only if the order $d$ of $b$ in $\mathbb{Z}_{n}^{*}$ divides $n - 1$.\\
(2) If $n$ is pseudo-prime to the bases $b_{1}$ and $b_{2}$ such that gcd($b_{1}, n$) = $1$ and gcd($b_{2}, n$) = $1$, then $n$ is also pseudo-prime to the bases $b_{1}, b_{2}, b_{1}b_{2}^{-1}$, and $b_{1}^{-1}b_{2}$.\\
(3) If there is a $b \in \mathbb{Z}_{n}^{*}$ satisfying $b_{n - 1} \not \equiv 1$ mod $n$, then
\begin{center}
$|\{b \in \mathbb{Z}_{n}^{*}| b^{n - 1} \not \equiv 1$ mod $n \} | \geq \frac{\varphi (n)}{2}$.
\end{center}
\end{theorem}
\end{samepage}


\begin{theorem}
Let $n \in \mathbb{N}$ be an odd composite number. Then we have \\
(1) If there is a square number $q^{2} > 1$ dividing $n$ then $n$ is not a Carmichael number. \\
(2) If $n$ is square-free, then $n$ is Carmichael number if and only if $(p - 1)$ divides $n - 1$ for every prime $p$ dividing $n$.
\end{theorem}



\section{Exercises}
\noindent \textbf{Exercise 5.1}  Let $p$ be a prime number. Then $\mathbb{Z}^{*}_{p^2}$ is cyclic.
\begin{proof}
We distinguish the cases $p = 2$ and $p > 2$.\\
Case $1$. $p = 2$\\
Since $\mathbb{Z}_{2^{2}}^{*} = \mathbb{Z}_{4}^{*} = \{1, 3 \}$, it suffices to test $3$. We compute $3^{1} \equiv 3$ mod $4$, as well as $3^2 = 9 \equiv 1$ mod $4$ and thus, $\mathbb{Z}_{2^{2}}^{*}$ is cyclic.\\
Case $2$. $p > 2$\\
By \textbf{Corollary 2 from Lecture 3} we know that $\mathbb{Z}_{p}^{*}$ is a cyclic group of order $p - 1$. Let $g \in \mathbb{Z}_{p}^{*}$ be a generator for $\mathbb{Z}_{p}^{*}$. Thus, it suffices to show the following claim:\\
Claim $1$. Either $g$ or $g + p$ is a generator for $\mathbb{Z}_{p^{2}}^{*}$.\\
Let $n$ be the order of $g$ in $\mathbb{Z}_{p^{2}}^{*}$. Then $g^{n} \equiv 1$ mod $p^{2}$, and consequently we also have $g^{n} \equiv 1$ mod $p$. So $p - 1$ must divide $n$. Moreover since $n$ is the order of $g$ in $\mathbb{Z}_{p^{2}}^{*}$, we also know that $n$ divides $\varphi (p^{2})$ (this comes from \textbf{Corollary 1 from Lecture 1}). By \textbf{Theorem 11 from Lecture 2} we know that $\varphi (p^{2}) = p(p - 1)$, and hence, $n|p(p - 1)$. Thus, we have both $(p - 1)|n$ and $n|p(p - 1)$, and consequently either $n = p - 1$ or $n = p(p - 1)$.\\
If $n = p(p - 1)$, then $g$ is also a generator for $\mathbb{Z}_{p^{2}}^{*}$, and we are done.\\
Now, assume $n = p - 1$. Therefore, we know that
$$g^{p - 1} \equiv 1\ mod\ p^{2}$$
Let $s =_{df} g + p$. Hence, $s \equiv g$ mod $p$, and so $s$ is also a generator for $\mathbb{Z}_{p}^{*}$. Using the same arguments as above we conclude that the order of $s$ in $\mathbb{Z}_{p^{2}}^{*}$ is either $p - 1$ or $p(p - 1)$. Thus it suffices to show that $s^{p - 1} \not \equiv 1$ mod $p^{2}$.\\
Using the binomial theorem we obtain
$$s^{p-1} = (g + p)^{p - 1} = \sum_{k = 0}^{p - 1} \binom{p-1}{k} g^{k}p^{p - 1 - k} \equiv g^{p - 2} \cdot p(p - 1) + g^{p - 1}\ mod\ p^{2}$$
From $g^{p - 1} \equiv 1$ mod $p^{2}$ we obtain:
$$s^{p - 1} \equiv 1 - pg^{p-2}\ mod\ p^{2}$$
Finally, suppose $s^{p - 1} \equiv 1$ mod $p^{2}$. Then we get $pg^{p - 2} \equiv 0$ mod $p^{2}$. But this means that there must be a $k$ such that $k \cdot p^{2} = pg^{p - 2}$, and consequently, $kp = g^{p - 2}$. Thus $g^{p - 2} \equiv 0$ mod $p$, a contradiction, since $g$ is a generator of $\mathbb{Z}_{p}^{*}$.\\
Taking this all together, we see that $s^{p - 1} \equiv 1 - pg^{p-2}$ mod $p^{2}$ implies $s^{p - 1} \not \equiv 1$ mod $p^{2}$, and so $s$ has order $p(p - 1)$. Hence $s$ is a generator for $\mathbb{Z}_{p^{2}}^{*}$. Consequently, $\mathbb{Z}_{p^{2}}^{*}$ is cyclic.
\end{proof}

\noindent \textbf{Exercise 5.2} Every Carmichael number is the product of at least 3 distinct primes.
\begin{proof}
We prove that no Carmichael number can be the product of exactly 2 distinct primes. Suppose, for contradiction, that $n = pq$ is a Carmichael number where $p$ and $q$ are distinct primes with $p > q$. By definition and Theorem 9(2), $n$ is square-free, so $p$ and $q$ are indeed distinct.

By Theorem 9(2), a square-free number $n$ is a Carmichael number if and only if $(p - 1) \mid (n - 1)$ for every prime $p$ dividing $n$. Applying this to $n = pq$:
\begin{center}
$(p - 1) \mid (pq - 1)$ \quad \text{and} \quad $(q - 1) \mid (pq - 1)$
\end{center}

From $(p - 1) \mid (pq - 1)$, there exists $k \in \mathbb{Z}$ such that $(p-1)k = pq - 1$. Also, $(p - 1) \mid pq(p-1)$ trivially, which gives us $(p-1)(q-1) \mid (pq-1)$. But since $pq - 1 = (p-1)q + (q-1)$, we have:
\begin{center}
$(p-1) \mid (pq - 1 - (p-1)q) = (q - 1)$
\end{center}

This means $(p-1) \mid (q-1)$. However, since $p$ and $q$ are distinct primes with $p > q \geq 2$, we have $p \geq q + 1$, so $p - 1 \geq q > q - 1$. Therefore, $(p-1) > (q-1)$, and it is impossible for $(p-1)$ to divide a smaller positive number $(q-1)$ unless $q - 1 = 0$, which contradicts $q$ being a prime $\geq 2$.

Thus, $n$ cannot be a Carmichael number if $n = pq$ for primes $p, q$. Therefore, every Carmichael number must be the product of at least 3 distinct primes.
\end{proof}

\noindent \textbf{Exercise 5.3} Prove equation (2) in \textbf{Example 5.1}.
\begin{proof}
The equation (2) from the said example is as follows:
\begin{center}
$x(x + 1)(x^3 + x + 1)(x^3 + x^2 + 1) = x^8 - x$
\end{center}
over $\mathbb{Z}_{2}$. In $\mathbb{Z}_2$, we have $1 + 1 = 0$, which implies $-1 \equiv 1 \pmod{2}$ and $2x \equiv 0 \pmod{2}$ for any $x$.

We expand the left-hand side step by step:
\begin{center}
$x(x + 1)(x^3 + x + 1)(x^3 + x^2 + 1)$
\end{center}

First, multiply $x(x + 1) = x^2 + x$. Then multiply by $(x^3 + x + 1)$:
\begin{center}
$(x^2 + x)(x^3 + x + 1) = x^5 + x^3 + x^2 + x^4 + x^2 + x = x^5 + x^4 + x^3 + x$
\end{center}
where we used $x^2 + x^2 = 2x^2 \equiv 0 \pmod{2}$ to cancel like terms.

Next, multiply by $(x^3 + x^2 + 1)$:
\begin{center}
$(x^5 + x^4 + x^3 + x)(x^3 + x^2 + 1)$
\end{center}

Expanding:
\begin{center}
$= x^8 + x^7 + x^5 + x^7 + x^6 + x^4 + x^6 + x^5 + x^3 + x^4 + x^3 + x$
\end{center}

Now, collecting terms and using $2y \equiv 0 \pmod{2}$ for any $y$, the terms that appear twice cancel:
- $x^7$ appears twice: $x^7 + x^7 \equiv 0$
- $x^6$ appears twice: $x^6 + x^6 \equiv 0$
- $x^5$ appears twice: $x^5 + x^5 \equiv 0$
- $x^4$ appears twice: $x^4 + x^4 \equiv 0$
- $x^3$ appears twice: $x^3 + x^3 \equiv 0$

Therefore, we are left with:
\begin{center}
$x^8 + x \equiv x^8 - x \pmod{2}$
\end{center}
where the last equality holds because $-x \equiv x \pmod{2}$ (since $-1 \equiv 1 \pmod{2}$). Thus, the equation is proven.
\end{proof}

\noindent \textbf{Exercise 5.4} Calculate $5^{71}$ mod 31 and provide your solution in the way of the example given at page 39.
\begin{proof}
We use binary exponentiation. First, we write 71 in binary: $71 = 64 + 4 + 2 + 1 = 2^6 + 2^2 + 2^1 + 2^0$, so $71 = (1000111)_2$ in binary.

We compute successive squares of 5 modulo 31:
\begin{center}
$y_0 = 5^{2^0} = 5^1 \equiv 5 \pmod{31}$\\
$y_1 = 5^{2^1} = 5^2 \equiv 25 \pmod{31}$\\
$y_2 = 5^{2^2} = 5^4 \equiv 25^2 = 625 \equiv 625 - 20 \cdot 31 = 625 - 620 = 5 \pmod{31}$\\
$y_3 = 5^{2^3} = 5^8 \equiv 5^2 = 25 \pmod{31}$\\
$y_4 = 5^{2^4} = 5^{16} \equiv 25^2 = 625 \equiv 5 \pmod{31}$\\
$y_5 = 5^{2^5} = 5^{32} \equiv 5^2 = 25 \pmod{31}$\\
$y_6 = 5^{2^6} = 5^{64} \equiv 25^2 = 625 \equiv 5 \pmod{31}$
\end{center}

Since $71 = (1000111)_2$, we have $5^{71} = 5^{64} \cdot 5^4 \cdot 5^2 \cdot 5^1$. Using the computed values:
\begin{center}
$5^{71} \equiv y_6 \cdot y_2 \cdot y_1 \cdot y_0 \equiv 5 \cdot 5 \cdot 25 \cdot 5 \pmod{31}$
\end{center}

Computing step by step:
\begin{center}
$5 \cdot 5 = 25$\\
$25 \cdot 25 = 625 \equiv 5 \pmod{31}$\\
$5 \cdot 5 = 25$
\end{center}

Therefore, $5^{71} \equiv 25 \pmod{31}$.

\textbf{Verification using Fermat's Little Theorem:} Since 31 is prime, by Fermat's Little Theorem, $5^{30} \equiv 1 \pmod{31}$. We can verify by computing $71 = 2 \cdot 30 + 11$, so $5^{71} = (5^{30})^2 \cdot 5^{11} \equiv 5^{11} \pmod{31}$. Computing $5^{11} = 5^8 \cdot 5^2 \cdot 5 \equiv 25 \cdot 25 \cdot 5 = 625 \cdot 5 \equiv 5 \cdot 5 = 25 \pmod{31}$, which confirms our result.
\end{proof}

\noindent \textbf{Exercise 5.5} Prove Assertion (2) of \textbf{Theorem 8}. The assertion reads:
\begin{center}
If $n$ is pseudo-prime to the bases $b_1$ and $b_2$ such that gcd($b_1$, $n$) = 1 and gcd($b_2$, $n$) = 1, then $n$ is also pseudo-prime to the bases $b_{1}b_{2}$, $b_{1}b_{2}^{-1}$, and $b_{1}^{-1}b_{2}$.
\end{center}
\begin{proof}
By definition, $n$ is pseudo-prime to base $b$ if $b^{n-1} \equiv 1 \pmod{n}$. Given that $n$ is pseudo-prime to bases $b_1$ and $b_2$, we have:
$$b_1^{n-1} \equiv 1 \pmod{n} \quad \text{and} \quad b_2^{n-1} \equiv 1 \pmod{n}$$

\textbf{For base $b_1 b_2$:}
\begin{center}
$(b_1 b_2)^{n-1} = b_1^{n-1} b_2^{n-1} \equiv 1 \cdot 1 = 1 \pmod{n}$
\end{center}
Therefore, $n$ is pseudo-prime to base $b_1 b_2$.

\textbf{For base $b_1^{-1}$:}
From $b_1^{n-1} \equiv 1 \pmod{n}$, we can take the inverse of both sides:
\begin{center}
$(b_1^{n-1})^{-1} \equiv 1^{-1} \pmod{n}
\end{center}
This gives us $(b_1^{-1})^{n-1} \equiv 1 \pmod{n}$, so $n$ is pseudo-prime to base $b_1^{-1}$.

\textbf{For base $b_2^{-1}$:}
Similarly, from $b_2^{n-1} \equiv 1 \pmod{n}$, we obtain $(b_2^{-1})^{n-1} \equiv 1 \pmod{n}$, so $n$ is pseudo-prime to base $b_2^{-1}$.

\textbf{For base $b_1 b_2^{-1}$:}
Since $n$ is pseudo-prime to bases $b_1$ and $b_2^{-1}$:
\begin{center}
$(b_1 b_2^{-1})^{n-1} = b_1^{n-1} (b_2^{-1})^{n-1} \equiv 1 \cdot 1 = 1 \pmod{n}$
\end{center}
Therefore, $n$ is pseudo-prime to base $b_1 b_2^{-1}$.

\textbf{For base $b_1^{-1} b_2$:}
Since $n$ is pseudo-prime to bases $b_1^{-1}$ and $b_2$:
\begin{center}
$(b_1^{-1} b_2)^{n-1} = (b_1^{-1})^{n-1} b_2^{n-1} \equiv 1 \cdot 1 = 1 \pmod{n}$
\end{center}
Therefore, $n$ is pseudo-prime to base $b_1^{-1} b_2$.

This completes the proof of Assertion (2).
\end{proof}
