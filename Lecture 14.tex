A process $i$ (for $i = 1, \ldots, |V|$) consists formally of the following components: \\
\begin{itemize}
\item $\operatorname{states}_{i}$: a possibly infinite set of states
\item $\operatorname{start}_{i}$: a non-empty set of start states, which is a subset of $\operatorname{states}_{i}$
\item $\operatorname{msg}_{i}$: a message-generation function depending only on $\operatorname{states}_{i}$ and the edges outgoing from node $i$
\item $\operatorname{trans}_{i}$: a state transition function mapping $\operatorname{states}_{i} \times (M^* \cup \{\lambda\})$ to $\operatorname{states}_{i}$, where $M^*$ denotes the set of all finite sequences of messages and $\lambda$ denotes the empty sequence
\end{itemize}
\begin{theorem}
Let $n \in \mathbb{N}, n > 1$, and let $A$ be a system of $n$ processes arranged in a bidirectional ring. If all processes in $A$ are identical, then $A$ cannot solve the leader election problem.
\end{theorem}


\begin{theorem}
There exists an algorithm LE solving the leader election problem in time $n$ and with communication complexity $O(n^{2})$ for every system $A$ of $n$ processes arranged in a ring, where $n \in \mathbb{N}, n > 1$, is not known to the algorithm. Moreover, LE uses only unidirectional communication.
\end{theorem}


\begin{theorem}
\textbf{Hirschberg and Sinclair Theorem -} There exists an algorithm HS solving the leader election problem in time $O(n)$ and with communication complexity $O(n \log{n})$ for every system $A$ of $n$ processes arranged in a ring, where $n \in \mathbb{N}, n > 1$, is not known to the algorithm.
\end{theorem}

\begin{algorithm}
\textbf{Hirschberg-Sinclair Algorithm HS (Leader Election in Ring)} \\
Each process $i$ (for $i = 1, \ldots, n$) executes the following algorithm:

\textbf{Initialization:} Each process $i$ initializes its UID to $u_i$ and sends tokens in both directions.

\textbf{Phase $l$ (for $l = 0, 1, 2, \ldots$):} 

Each process $i$ maintains a triple $(u, b, k)$ where:
\begin{itemize}
\item $u$ is the UID being tested
\item $b \in \{\operatorname{out}, \operatorname{in}\}$ indicates message direction (out = being tested, in = returning)
\item $k$ is the hop counter
\end{itemize}

\textbf{Outgoing phase:} Process $i$ initializes a token with $(u_i, \operatorname{out}, 2^l)$ and sends copies in both clockwise and counter-clockwise directions. Set $\operatorname{status} = \operatorname{unknown}$ and $t = 0$ (counter for returning tokens).

\textbf{Processing incoming messages:}

(1) \textit{If message $v = (u_j, \operatorname{in}, k)$ arrives (return phase):}
\begin{itemize}
\item If $u_j = u_i$: Increment $t := t + 1$. If $t = 2$, both returning tokens have arrived; stop Phase $l$ and start Phase $l+1$.
\item If $u_j \neq u_i$: Forward the token in the direction it came from (send back).
\end{itemize}

(2) \textit{If message $v = (u_j, \operatorname{out}, k)$ arrives (outgoing phase):}
\begin{itemize}
\item If $u_j > u_i$: If $k > 1$, decrement the counter and forward: send $(u_j, \operatorname{out}, k-1)$. If $k = 1$, change direction and return: send $(u_j, \operatorname{in}, 0)$.
\item If $u_j < u_i$: Drop the message (do nothing).
\item If $u_j = u_i$: Set $\operatorname{status} = \operatorname{leader}$ (this process has the unique maximum UID).
\end{itemize}

\textbf{Termination:} When a process reaches status = leader, it has won the election.

\textit{Correctness:} The algorithm ensures that only the process with the globally unique maximum UID will eventually set its status to leader. The proof follows by analyzing the propagation of tokens with decreasing hop counts and the filtering property that drops tokens from processes with smaller UIDs.
\end{algorithm}
