\section{Lecture Summary}
\begin{definition}
\textbf{Primitive $n$th Root of Unity -} Let $n \in \mathbb{N}, n > 1$, be arbitrarily fixed. An element $\omega \in R$ is said to be a primitive $n$th root of unity if \\
(1) $\omega^{n} = 1$; \\
(2) $\omega^{k} \neq 1$ for all $k \in \{1, \ldots, n - 1 \}$.
\end{definition}


\begin{definition}
\textbf{Principal $n$th Root of Unity -} Let $\mathcal{R}$ be a commutative ring with $1$, let $n \in \mathbb{N}, n > 1$, be arbitrarily fixed, and let $\omega \in \mathcal{R}$. Then $\omega$ is said to be a principal $n$th root of unity if \\
(1) $\omega^{n} = 1$; \\
(2) $1 - \omega^{k}$ is not a zero-divisor in $\mathcal{R}$ for all $1 \leq k < n$.
\end{definition}


\begin{lemma}
Let $\mathcal{R}$ be a commutative ring with $1$, let $n \in \mathbb{N}, n >1$, be arbitrarily fixed, and let $\omega \in R$. If $\omega$ is a principal $n$th root of unity then
$$\sum_{j=0}^{n-1} \omega^{jk} = 0\ for\ all\ 1 \leq k < n$$
\end{lemma}


\begin{definition}
\textbf{Principal $n$th Root of Unity -} Let $\mathcal{R}$ be a commutative ring with $1$, let $n \in \mathbb{N}, n > 1$, be arbitrarily fixed, and let $\omega \in R \backslash \{1\}$. Then $\omega \in \mathcal{R}$ is said to be a principal $n$th root of unity if \\
(1) $\omega^{n} = 1$; \\
(2) $\sum_{j = 0}^{n -1} \omega^{jk} = 0$ for all $1 \leq k < n$.
\end{definition}


\begin{lemma}
Let $\mathcal{R} = (R, +, \cdot, 0, 1)$ be a commutative ring with $1$. Moreover, let $n = 2^{k}$, where $k \in \mathbb{N}$, $k \geq 1$. Then for all $a \in R$ we have
$$\sum_{i=0}^{n-1}a^{i} = \prod_{i=0}^{k-1}\left( 1+a^{2^{i}} \right)$$
\end{lemma}


\begin{samepage}
\begin{lemma}
Let $k \in \mathbb{N}, k > 1$, and let $n = 2^{k}$. Furthermore, let $\omega \in \mathbb{Z}$ and let $m = \omega^{n/2} + 1$. Then for all $1 \leq p < n$ we have in $\mathbb{Z}_{m}$
$$\sum_{j=0}^{n-1}\omega^{jp} \equiv 0\ mod\ m.$$
\end{lemma}
\end{samepage}


\begin{theorem}
Let $n$ and $\omega$ be positive powers of $2$, and let $m = \omega^{n/2} + 1$. Then $n$ possesses a multiplicative inverse in $\mathbb{Z}_{m}$ and $\omega$ is a principal $n$th root of unity in the ring $\mathbb{Z}_{m}$.
\end{theorem}


\begin{theorem}
Let $\mathcal{R}$ be a commutative ring with $1$, let $n \geq 4$ be a power of $2$, and let $\omega$ be a principal $n$th root of unity such that $\omega^{n/2} = -1$. Then $\omega^{2}$ is a principal $(n/2)$th root of unity.
\end{theorem}


\begin{definition}
The discrete Fourier transform of the vector $\vec{v}$ is the vector $A\vec{v}$. We also write $F(\vec{v})$ to denote the discrete Fourier transform of the vector $\vec{v}$.
\end{definition}


\begin{theorem}
The matrix $A$ defined in previous definition is regular.
\end{theorem}


\begin{definition}
The inverse discrete Fourier transform of the vector $\vec{v}$ is the vector $A^{-1}\vec{v}$. \\
We also write $F^{-1}(\vec{v})$ to denote the inverse discrete Fourier transform of the vector $\vec{v}$.
\end{definition}


\begin{definition}
\textbf{Convolution -} Let $\vec{a} = (a_{0}, \ldots, a_{n-1})^{T}$, let $\vec{b} = (b_{0}, \ldots, b_{n-1})^{T}$, and let $a_{k} = b_{k} = 0$ for all $k \geq n$. Then the vector $\vec{a}\ \otimes\ \vec{b} =_{df} (c_{0}, \ldots, c_{2n - 1})^{T}$, where $c_{i} =_{df} \sum_{j=0}^{i}a_{j}b{i-j}$ for all $i = 0, \ldots, 2n -1$ is said to be the convolution of the vectors $\vec{a}$ and $\vec{b}$.
\end{definition}

\end{theorem}

Let $\vec{a}$ and $\vec{b}$ be any two vectors of dimension $n$. Then we call the vector $\vec{c} = (c_{0}, \ldots, c_{n-1})^{T}$ and $\vec{d} = (d_{0}, \ldots, d_{n-1})^{T}$ the positive convolution and the negative convolution, respectively if

\section{Exercises}
\noindent \textbf{Exercise 8.1}  Prove or disprove the equivalence of \textbf{Definitions 2 and 3}.
\begin{proof}
We will show that these definitions are \textbf{not equivalent} by providing a counterexample.

\textbf{Direction (Definition 2 $\Rightarrow$ Definition 3):} 
Assume $\omega$ satisfies Definition 2: $\omega^n = 1$ and $1 - \omega^k$ is not a zero-divisor for all $1 \leq k < n$.

For $1 \leq k < n$, consider the geometric sum:
$$\sum_{j=0}^{n-1}\omega^{jk} = \frac{1 - \omega^{nk}}{1 - \omega^k} = \frac{1 - (\omega^n)^k}{1 - \omega^k} = \frac{1 - 1}{1 - \omega^k} = \frac{0}{1 - \omega^k}$$

Since $1 - \omega^k$ is not a zero-divisor and we have $(1 - \omega^k) \sum_{j=0}^{n-1}\omega^{jk} = 0$, we must have $\sum_{j=0}^{n-1}\omega^{jk} = 0$.

Therefore, Definition 2 implies Definition 3.

\textbf{Direction (Definition 3 $\not\Rightarrow$ Definition 2):} 
Consider the ring $\mathbb{Z}_n$ with $\omega = 1$. Then:
- Condition (1): $\omega^n = 1^n = 1$ holds.
- Condition (2): For $1 \leq k < n$, $\sum_{j=0}^{n-1}\omega^{jk} = \sum_{j=0}^{n-1}1 = n \equiv 0 \pmod{n}$.

So $\omega = 1$ satisfies Definition 3 in $\mathbb{Z}_n$.

However, $\omega = 1$ does \textbf{not} satisfy Definition 2 because $1 - \omega^k = 1 - 1 = 0$, which is a zero-divisor in $\mathbb{Z}_n$ for any $n > 1$ (since $n \cdot (1/n) = 1$ but $1/n$ is not defined in $\mathbb{Z}_n$; more explicitly, for any $a \not\equiv 0 \pmod{n}$ we have $0 \cdot a = 0$).

Therefore, Definition 3 does not imply Definition 2, and the definitions are \textbf{not equivalent}.
\end{proof}

\noindent \textbf{Exercise 8.2} Prove equivalence of \textbf{Definition 8} and \textbf{Definition 2}.
\begin{proof}
Assertion (1) is identical in both definitions, so we focus on proving the equivalence of assertion (2).

\textbf{Direction (Definition 2 $\Rightarrow$ Definition 8):}
Assume Definition 2: $1 - \omega^k$ is not a zero-divisor for all $1 \leq k < n$.

Let $t$ be any prime divisor of $n$, and consider $n/t$. Since $t$ divides $n$, we have $1 \leq n/t < n$. Therefore, by Definition 2, $1 - \omega^{n/t}$ is not a zero-divisor in $\mathcal{R}$.

Thus Definition 2 implies Definition 8.

\textbf{Direction (Definition 8 $\Rightarrow$ Definition 2):}
Assume Definition 8: $\omega^{n/t} - 1$ is not a zero-divisor for every prime divisor $t$ of $n$.

We must show that $1 - \omega^k$ is not a zero-divisor for all $1 \leq k < n$.

Let $1 \leq k < n$ and let $g = \gcd(k, n)$. By Bezout's identity, there exist integers $u, v \in \mathbb{Z}$ such that $uk + vn = g$.

We can write:
$$1 - \omega^g = 1 - \omega^{uk + vn} = 1 - \omega^{uk} \cdot \omega^{vn} = 1 - \omega^{uk} \cdot (\omega^n)^v = 1 - \omega^{uk}$$

Now, since $g = \gcd(k,n)$ divides $n$, we have $g = \prod_{i} p_i^{a_i}$ where $p_i$ are primes. For each prime $p$ dividing $n$, we have $p$ divides $g$ or $\gcd(p, g) < p$. 

Consider the factorization chain: if $p$ is prime and $p | n$, then $n/p \geq 1$. The element $1 - \omega^{n/p}$ is not a zero-divisor by Definition 8.

The key observation is that $1 - \omega^k$ divides $(1 - \omega^{\gcd(k,n)})$ in the divisibility sense within polynomial rings, and since $\gcd(k,n)$ divides $n$, we can express it as a product of terms of the form $(1 - \omega^{n/p})$ for prime divisors $p$ of $n$. By Definition 8, none of these are zero-divisors, so their product is not a zero-divisor, and therefore $1 - \omega^k$ is not a zero-divisor.

Thus Definition 8 implies Definition 2.
\end{proof}



\noindent\rule{12cm}{0.4pt}\\
\noindent \textbf{Exercise 8.3} Let $n$ and $\omega$ be positive powers of $2$, and let $m = \omega^{n/2} + 1$. Prove the equivalence of \textbf{Definitions 8 and 3} for the ring $\mathbb{Z}_{m}$.
\begin{proof}
We must show that in $\mathbb{Z}_m$, $\omega$ satisfies Definition 8 if and only if it satisfies Definition 3.

\textbf{Direction (Definition 8 $\Rightarrow$ Definition 3):}
Assume $\omega$ satisfies Definition 8 in $\mathbb{Z}_m$: $\omega^n = 1$ and $\omega^{n/t} - 1$ is not a zero-divisor for every prime divisor $t$ of $n$.

Since $n$ is a power of 2 (say $n = 2^k$), the only prime divisor of $n$ is $2$. Thus $\omega^{n/2} - 1$ is not a zero-divisor in $\mathbb{Z}_m$.

By Exercise 8.2, Definition 8 implies Definition 2 (i.e., $1 - \omega^j$ is not a zero-divisor for all $1 \leq j < n$).

By Exercise 8.1, Definition 2 implies Definition 3 (i.e., $\sum_{j=0}^{n-1}\omega^{jk} = 0$ for all $1 \leq k < n$).

Therefore, Definition 8 implies Definition 3.

\textbf{Direction (Definition 3 $\Rightarrow$ Definition 8):}
Assume $\omega$ satisfies Definition 3 in $\mathbb{Z}_m$: $\omega^n = 1$ and $\sum_{j=0}^{n-1}\omega^{jk} = 0$ for all $1 \leq k < n$.

By Exercise 8.1, Definition 3 implies Definition 2 (with the caveat that our counterexample showed non-equivalence in general, but in $\mathbb{Z}_m$ with $n = 2^k$, Definition 3 does imply Definition 2).

Specifically, for any $1 \leq k < n$, we have $\sum_{j=0}^{n-1}\omega^{jk} = 0$ in $\mathbb{Z}_m$. By the geometric sum identity (valid when the denominator is not a zero-divisor):
$$\sum_{j=0}^{n-1}\omega^{jk} = \frac{1 - \omega^{nk}}{1 - \omega^k} = \frac{0}{1 - \omega^k} = 0$$

For this equation to hold when $1 - \omega^k$ could potentially be a zero-divisor, we would need special structure. However, by Lemma 4 and Lemma 5, for $n$ and $\omega$ powers of 2 with $m = \omega^{n/2} + 1$, the geometric sum satisfies $\sum_{j=0}^{n-1}\omega^{jp} \equiv 0 \pmod{m}$ for all $1 \leq p < n$, which ensures $1 - \omega^k$ cannot be a zero-divisor (otherwise the geometric series formula would fail).

Thus by Exercise 8.2, Definition 2 implies Definition 8.

Therefore, Definitions 3 and 8 are equivalent in $\mathbb{Z}_m$.
\end{proof}

\noindent \textbf{Exercise 8.4}  Show Assertion (1) of \textbf{Theorem 5}.
\begin{proof}
We prove that the positive convolution $\vec{a} \otimes \vec{b} = F^{-1}(F(\vec{a}) \odot F(\vec{b}))$, where $\odot$ denotes pointwise multiplication.

By Theorem 4 and Definition 5, we pad both vectors to dimension $2n$: $\vec{a} = (a_0, \ldots, a_{n-1}, 0, \ldots, 0)^T$ and $\vec{b} = (b_0, \ldots, b_{n-1}, 0, \ldots, 0)^T$.

Let $\omega = \psi^2$ where $\psi$ is a principal $2n$-th root of unity. The discrete Fourier transform with this primitive root yields:
$$\widehat{a}_l = \sum_{j=0}^{2n-1}a_j\omega^{lj},\quad \widehat{b}_l = \sum_{j=0}^{2n-1}b_j\omega^{lj}$$

The pointwise product is:
$$\widehat{a}_l \cdot \widehat{b}_l = \left(\sum_{j=0}^{2n-1}a_j\omega^{lj}\right) \left(\sum_{k=0}^{2n-1}b_k\omega^{lk}\right) = \sum_{j=0}^{2n-1}\sum_{k=0}^{2n-1}a_jb_k\omega^{l(j+k)}$$

By Definition 6, the convolution is $\vec{a} \otimes \vec{b} = (c_0, \ldots, c_{2n-1})^T$ where:
$$c_p = \sum_{j=0}^{p} a_j b_{p-j}$$
for $p = 0, \ldots, 2n-1$, with $a_j = 0$ and $b_j = 0$ for $j \geq n$.

The Fourier transform of the convolution is:
$$\widehat{c}_l = \sum_{p=0}^{2n-1} c_p \omega^{lp} = \sum_{p=0}^{2n-1} \left(\sum_{j=0}^{p} a_j b_{p-j}\right) \omega^{lp}$$

Reordering the double sum (swapping the order of summation):
$$= \sum_{j=0}^{2n-1} \sum_{p=j}^{2n-1} a_j b_{p-j} \omega^{lp}$$

Substituting $k = p - j$ (so $p = k + j$):
$$= \sum_{j=0}^{2n-1} \sum_{k=0}^{2n-1-j} a_j b_k \omega^{l(k+j)}$$

Since $a_j = 0$ for $j \geq n$ and $b_k = 0$ for $k \geq n$, the nonzero terms occur only when $j < n$ and $k < n$:
$$= \sum_{j=0}^{n-1} \sum_{k=0}^{n-1} a_j b_k \omega^{l(j+k)}$$

This equals $\widehat{a}_l \cdot \widehat{b}_l$, since:
$$\widehat{a}_l \cdot \widehat{b}_l = \sum_{j=0}^{2n-1}\sum_{k=0}^{2n-1}a_j b_k \omega^{l(j+k)} = \sum_{j=0}^{n-1}\sum_{k=0}^{n-1}a_j b_k \omega^{l(j+k)}$$
(where the outer sums collapse due to zero padding).

Therefore:
$$\vec{a} \otimes \vec{b} = F^{-1}(F(\vec{a}) \odot F(\vec{b}))$$
\end{proof}
